"use client";

import { useState, type CSSProperties } from "react";
import { Segmented } from "@/components/creative-technology/ui/segmented";

type Sample = { name: string; tilt: number; scale: number; x: number; y: number; route?: "review" | "rejected" };

/**
 * A model of headshot normalization. Every tile starts as it was supplied:
 * tilted, too near or too far, off centre. Normalizing levels each face and
 * puts every pair of eyes on the same line at the same size.
 *
 * It is a diagram, not output: it uses no photographs. Tiles the system
 * would set aside keep their place and say why.
 */
export function Normalizer({ batch }: { batch: Sample[] }) {
  const [state, setState] = useState("supplied");
  const normalized = state === "normalized";

  return (
    <>
      <div className="stage p-3 sm:p-6">
        <ul className="grid grid-cols-4 gap-2 sm:gap-3">
          {batch.map((sample) => {
            const held = Boolean(sample.route);
            const settle = normalized && !held;
            return (
              <li key={sample.name} className="relative aspect-square overflow-hidden rounded-s bg-white">
                {/* The line every pair of eyes should land on. */}
                <span className="absolute inset-x-0 top-[42%] h-px bg-line-strong" aria-hidden="true" />
                <span
                  aria-hidden="true"
                  className="absolute left-1/2 top-[42%] block h-[46%] w-[46%] transition-transform duration-slow ease-out"
                  style={
                    {
                      transform: settle
                        ? "translate(-50%, -38%)"
                        : `translate(calc(-50% + ${sample.x * 100}%), calc(-38% + ${sample.y * 100}%)) rotate(${sample.tilt}deg) scale(${sample.scale})`,
                    } as CSSProperties
                  }
                >
                  <span className={`absolute inset-0 rounded-full border ${sample.route === "rejected" ? "border-dashed border-ink-3" : "border-ink"}`} />
                  {sample.route === "rejected" ? null : <span className="absolute inset-x-[18%] top-[38%] h-px bg-ink" />}
                </span>
                {held && normalized ? (
                  <span className="absolute inset-x-0 bottom-1 text-center text-[0.625rem] leading-none text-ink-2 sm:bottom-1.5 sm:text-[0.6875rem]">
                    {sample.route === "review" ? "Review" : "No face"}
                  </span>
                ) : null}
              </li>
            );
          })}
        </ul>
      </div>

      <div className="controls">
        <Segmented
          label="Batch state"
          value={state}
          onChange={setState}
          options={[
            { key: "supplied", label: "As supplied" },
            { key: "normalized", label: "Normalized" },
          ]}
        />
      </div>
    </>
  );
}
